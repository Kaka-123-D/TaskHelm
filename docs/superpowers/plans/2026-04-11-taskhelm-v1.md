# TaskHelm V1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a local-first AI engineering manager that lets one operator manage multiple projects, tasks, branches, worktrees, agents, and dev servers from CLI and web dashboard.

**Architecture:** Monorepo with shared TypeScript core. SQLite for runtime state (WAL mode, plain SQL migrations). Markdown/YAML on disk for human-readable task memory. Local supervisor daemon for event-driven orchestration. Next.js web dashboard + CLI as equal first-class interfaces.

**Tech Stack:** TypeScript, Node.js, Next.js 15 (App Router), SQLite (better-sqlite3), Commander.js (CLI), Turborepo (monorepo), Vitest (testing), Playwright (E2E), Tailwind CSS + shadcn/ui (dashboard)

```typescript
describe('runMigrations', () => {
  it('creates all expected tables', () => {
    const db = createDatabase(TEST_DB)

      .prepare("SELECT name FROM sqlite_master WHERE type='table' ORDER BY name")
```

```typescript
// packages/core/src/db/migrate.ts
import type Database from 'better-sqlite3'
import fs from 'node:fs'
import path from 'node:path'
import { fileURLToPath } from 'node:url'

const MIGRATIONS_DIR = path.join(
  path.dirname(fileURLToPath(import.meta.url)),
  'migrations'
)

export function runMigrations(db: Database.Database): void {
  db.exec(`
    CREATE TABLE IF NOT EXISTS _migrations (
      name TEXT PRIMARY KEY,
      applied_at TEXT NOT NULL
    )
  `)

  const applied = new Set(
    db.prepare('SELECT name FROM _migrations').all().map((r: any) => r.name)
  )

  const files = fs
    .readdirSync(MIGRATIONS_DIR)
    .filter((f) => f.endsWith('.sql'))
    .sort()

  const insert = db.prepare(
    'INSERT INTO _migrations (name, applied_at) VALUES (?, ?)'
  )

  for (const file of files) {
    if (applied.has(file)) continue
    const sql = fs.readFileSync(path.join(MIGRATIONS_DIR, file), 'utf-8')
    db.exec(sql)
    insert.run(file, new Date().toISOString())
  }
}
```

```typescript
// packages/core/src/models/project.ts
import type Database from 'better-sqlite3'
import { nanoid } from 'nanoid'
import type { Project } from '../types.js'

export interface CreateProjectInput {
  readonly name: string
  readonly slug: string
  readonly local_repo_root: string
  readonly description?: string
  readonly default_branch?: string
  readonly branch_naming_pattern?: string
  readonly worktree_root?: string
  readonly dev_command?: string
  readonly install_command?: string
  readonly test_command?: string
  readonly max_active_dev_servers?: number
  readonly specdown_mode?: string
  readonly specdown_project_ref?: string
}

export interface UpdateProjectInput {
  readonly name?: string
  readonly description?: string
  readonly default_branch?: string
  readonly branch_naming_pattern?: string
  readonly worktree_root?: string
  readonly dev_command?: string
  readonly install_command?: string
  readonly test_command?: string
  readonly max_active_dev_servers?: number
  readonly specdown_mode?: string
  readonly specdown_project_ref?: string
}

export class ProjectRepository {
  constructor(private readonly db: Database.Database) {}

  create(input: CreateProjectInput): Project {
    const now = new Date().toISOString()
    const id = nanoid()
    this.db
      .prepare(
        `INSERT INTO projects (
          id, slug, name, description, local_repo_root,
          default_branch, branch_naming_pattern, worktree_root,
          dev_command, install_command, test_command,
          max_active_dev_servers, specdown_mode, specdown_project_ref,
          created_at, updated_at
        ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`
      )
      .run(
        id,
        input.slug,
        input.name,
        input.description ?? null,
        input.local_repo_root,
        input.default_branch ?? null,
        input.branch_naming_pattern ?? null,
        input.worktree_root ?? null,
        input.dev_command ?? null,
        input.install_command ?? null,
        input.test_command ?? null,
        input.max_active_dev_servers ?? 1,
        input.specdown_mode ?? 'disabled',
        input.specdown_project_ref ?? null,
        now,
        now
      )
    return this.findById(id)!
  }

  findById(id: string): Project | null {
    return (this.db.prepare('SELECT * FROM projects WHERE id = ?').get(id) as Project) ?? null
  }

  findBySlug(slug: string): Project | null {
    return (this.db.prepare('SELECT * FROM projects WHERE slug = ?').get(slug) as Project) ?? null
  }

  findAll(): readonly Project[] {
    return this.db.prepare('SELECT * FROM projects ORDER BY created_at DESC').all() as Project[]
  }

  update(id: string, input: UpdateProjectInput): Project {
    const existing = this.findById(id)
    if (!existing) throw new Error(`Project not found: ${id}`)

    const now = new Date().toISOString()
    this.db
      .prepare(
        `UPDATE projects SET
          name = ?, description = ?, default_branch = ?,
          branch_naming_pattern = ?, worktree_root = ?,
          dev_command = ?, install_command = ?, test_command = ?,
          max_active_dev_servers = ?, specdown_mode = ?,
          specdown_project_ref = ?, updated_at = ?
        WHERE id = ?`
      )
      .run(
        input.name ?? existing.name,
        input.description ?? existing.description,
        input.default_branch ?? existing.default_branch,
        input.branch_naming_pattern ?? existing.branch_naming_pattern,
        input.worktree_root ?? existing.worktree_root,
        input.dev_command ?? existing.dev_command,
        input.install_command ?? existing.install_command,
        input.test_command ?? existing.test_command,
        input.max_active_dev_servers ?? existing.max_active_dev_servers,
        input.specdown_mode ?? existing.specdown_mode,
        input.specdown_project_ref ?? existing.specdown_project_ref,
        now,
        id
      )
    return this.findById(id)!
  }

  delete(id: string): void {
    this.db.prepare('DELETE FROM projects WHERE id = ?').run(id)
  }
}
```
